const btnCsv = document.getElementById('btn-export-csv');
    const btnExcel = document.getElementById('btn-export-excel');
    
    document.getElementById('export-periodo').addEventListener('change', (e) => {
      document.getElementById('export-custom-month').classList.toggle('hidden', e.target.value !== 'mes_especifico');
      document.getElementById('export-custom-date').classList.toggle('hidden', e.target.value !== 'fecha_especifica');
    });
    
    const getCustomVal = () => {
      const p = document.getElementById('export-periodo').value;
      if (p === 'mes_especifico') return document.getElementById('export-custom-month').value;
      if (p === 'fecha_especifica') return document.getElementById('export-custom-date').value;
      return null;
    };
    
    const generatedatesForPeriod = (periodo, customVal) => {
      let dates = [];
      let now = new Date();
      now.setHours(now.getHours() - 6);
      if (periodo === 'semana') {
        let day = now.getDay();
        let startDiff = now.getDate() - day + (day === 0 ? -6 : 1);
        let startDate = new Date(now.getFullYear(), now.getMonth(), startDiff);
        for (let i = 0; i < 5; i++) {
          let d = new Date(startDate);
          d.setDate(startDate.getDate() + i);
          dates.push(d.toISOString().split('T')[0]);
        }
      } else if (periodo === 'mes' || periodo === 'mes_especifico') {
        let year = now.getFullYear();
        let month = now.getMonth();
        if (periodo === 'mes_especifico' && customVal) {
          const [y, m] = customVal.split('-');
          year = parseInt(y, 10);
          month = parseInt(m, 10) - 1;
        }
        let daysInMonth = new Date(year, month + 1, 0).getDate();
        for (let i = 1; i <= daysInMonth; i++) {
          let d = new Date(year, month, i);
          let dayOfWeek = d.getDay();
          if (dayOfWeek !== 0 && dayOfWeek !== 6) {
            const mStr = String(month + 1).padStart(2, '0');
            const dStr = String(i).padStart(2, '0');
            dates.push(`${year}-${mStr}-${dStr}`);
          }
        }
      } else if (periodo === 'fecha_especifica' && customVal) {
        const [y, m, d] = customVal.split('-');
        const dayOfWeek = new Date(y, parseInt(m, 10) - 1, d).getDay();
        if (dayOfWeek !== 0 && dayOfWeek !== 6) {
          dates.push(customVal);
        }
      }
      return dates;
    };
    
    const doExport = async (format) => {
       const btn = format === 'csv' ? btnCsv : btnExcel;
       const originalText = btn.innerHTML;
       btn.innerHTML = 'Generando...';
       btn.disabled = true;
       
       try {
         const grado = document.getElementById('export-grado').value;
         const grupo = document.getElementById('export-grupo').value;
         const periodo = document.getElementById('export-periodo').value;
         const customVal = getCustomVal();
         const data = await getExportAsistencias(grado, grupo, periodo, customVal);
         
         const dates = generatedatesForPeriod(periodo, customVal);
         
         let csvContent = 'NO.,NOMBRE Y APELLIDO,';
         dates.forEach(d => csvContent += `${d.split('-')[2]},`);
         csvContent += '\n';
         
         let htmlContent = '<meta charset="UTF-8"><table border="1"><tr><th>NO.</th><th>NOMBRE Y APELLIDO</th>';
         dates.forEach(d => htmlContent += `<th>${d.split('-')[2]}</th>`);
         htmlContent += '</tr>';
         
         data.forEach((a, index) => {
            csvContent += `${index + 1},"${a.nombre}",`;
            htmlContent += `<tr><td>${index + 1}</td><td>${a.nombre}</td>`;
            
            const statusMap = {};
            if (a.asistencias) {
              a.asistencias.forEach(as => {
                 statusMap[as.date] = as.status;
              });
            }
            
            dates.forEach(d => {
               const st = statusMap[d] ? statusMap[d].toLowerCase() : '';
               let icon = '';
               if (st === 'a tiempo') icon = '?';
               else if (st === 'falta') icon = '?';
               else if (st === 'retardo') icon = 'R';
               else if (st === 'justificado') icon = 'J';
               else icon = '-';
               
               csvContent += `${icon},`;
               htmlContent += `<td>${icon}</td>`;
            });
            
            csvContent += '\n';
            htmlContent += '</tr>';
         });
         htmlContent += '</table>';
         
         const blob = format === 'csv' 
           ? new Blob(["\uFEFF"+csvContent], { type: 'text/csv;charset=utf-8;' }) 
           : new Blob([htmlContent], { type: 'application/vnd.ms-excel' });
         
         const url = URL.createObjectURL(blob);
         const link = document.createElement("a");
         link.href = url;
         link.download = `Asistencias_${grado}Grado_${periodo}.${format === 'csv' ? 'csv' : 'xls'}`;
         document.body.appendChild(link);
         link.click();
         document.body.removeChild(link);
       } catch(e) {
         console.error(e);
         alert('Error al generar la exportación.');
       }
       
       btn.innerHTML = originalText;
       btn.disabled = false;
    };
    
    if (btnCsv) btnCsv.addEventListener('click', () => doExport('csv'));
    if (btnExcel) btnExcel.addEventListener('click', () => doExport('excel'));

    const btnViewList = document.getElementById('btn-view-list');
    const vistaContainer = document.getElementById('vista-asistencias');
    if (btnViewList) {
      btnViewList.addEventListener('click', async () => {

             `;
             
             dates.forEach(d => {
                const dayNum = d.split('-')[2];
                htmlContent += `<th class="py-2 px-2 border border-gray-300 font-bold text-gray-700 text-center w-8" title="${d}">${dayNum}</th>`;

             data.forEach((a, index) => {
                htmlContent += `
                  <tr class="hover:bg-gray-50">

                    <td class="py-2 px-3 border border-gray-300 font-medium text-gray-800">${a.nombre}</td>
                `;
                
                const statusMap = {};
                if (a.asistencias) {
                  a.asistencias.forEach(as => {

                   else if (st === 'justificado') icon = '<span class="text-blue-600 font-bold text-base">J</span>';
                   else icon = '<span class="text-gray-300">-</span>';

             htmlContent += '</tbody></table>';

           document.getElementById('vista-asistencias-container').classList.remove('hidden');
         } catch(e) {
           console.error(e);
           alert('Error al generar la vista.');
         }
         
         btnViewList.innerHTML = originalText;
         btnViewList.disabled = false;
      });
    }
